% ---- rubriques
\bloc{Prérequis}{%
  Connaissances élémentaires en informatique ; utilisation d'un système d'exploitation ; notions de
  programmation (variables, conditions, boucles) et compréhension des structures de données simples.}

\bloc{Court descriptif de l'EC }{%
  Ce cours présente les principes de conception, de mise en place et d'administration des bases de données
  relationnelles. Il aborde la modélisation des données, le modèle entité-association, la transformation
  vers le modèle relationnel, le langage SQL, la normalisation, les transactions, la sécurité et les
  stratégies de sauvegarde. Les travaux pratiques s'appuient sur des cas concrets de gestion d'informations.}

\bloc{Objectif général de l'EC}{%
  Être capable de concevoir, interroger, sécuriser et administrer une base de données relationnelle adaptée
  aux besoins d'une organisation.}

\bloc{Objectifs spécifiques (savoir / savoir-faire / savoir-être)}{%
  \listeavec{Savoir}{\begin{itemize}
    \item expliquer les concepts fondamentaux des bases de données et le rôle d'un système de gestion de bases de données ;
    \item décrire les modèles conceptuel, logique et physique des données ;
    \item connaître les principes des clés, des contraintes d'intégrité et de la normalisation ;
    \item comprendre les mécanismes de transactions, de concurrence et de contrôle d'accès.
  \end{itemize}}
  \listeavec{Savoir-faire}{\begin{itemize}
    \item construire un modèle entité-association à partir d'un cahier des charges ;
    \item transformer un modèle conceptuel en schéma relationnel ;
    \item créer et modifier des tables à l'aide du langage SQL ;
    \item écrire des requêtes de sélection, de jointure, d'agrégation et de mise à jour ;
    \item mettre en œuvre des sauvegardes, restaurations et règles simples de sécurité.
  \end{itemize}}
  \listeavec{Savoir-être}{\begin{itemize}
    \item travailler avec méthode et précision dans la structuration des données ;
    \item documenter les choix de conception et respecter la confidentialité des informations ;
    \item collaborer efficacement lors de la réalisation d'un projet de base de données.
  \end{itemize}}}

\bloc{Contenu de l'enseignement }{%
  \begin{description}[leftmargin=0pt,labelindent=0pt,style=unboxed,font=\bfseries,itemsep=3pt]
    \item[Chapitre 1 : Introduction aux bases de données.] Notion de donnée et d'information ; fichiers et bases de données ; rôles et fonctions d'un système de gestion de bases de données ; architecture générale.
    \item[Chapitre 2 : Modélisation conceptuelle.] Recueil des besoins ; entités, associations et attributs ; cardinalités ; contraintes ; construction d'un modèle entité-association.
    \item[Chapitre 3 : Modèle relationnel.] Relations, domaines et tuples ; schémas relationnels ; clés primaires et étrangères ; contraintes d'intégrité ; passage du modèle conceptuel au modèle relationnel.
    \item[Chapitre 4 : Normalisation des données.] Dépendances fonctionnelles ; première, deuxième et troisième formes normales ; réduction des redondances et prévention des anomalies.
    \item[Chapitre 5 : Langage SQL.] Création et modification des tables ; insertion, mise à jour et suppression ; requêtes SELECT ; filtres, tris, jointures, fonctions d'agrégation et sous-requêtes.
    \item[Chapitre 6 : Transactions et concurrence.] Notion de transaction ; propriétés ACID ; validation et annulation ; verrous et anomalies liées aux accès concurrents.
    \item[Chapitre 7 : Administration et sécurité.] Gestion des utilisateurs et privilèges ; sauvegarde et restauration ; indexation ; notions de performance et de maintenance.
    \item[Chapitre 8 : Projet pratique et évaluation.] Conception et réalisation d'une base de données pour un cas d'étude ; rédaction de requêtes ; présentation et correction du projet.
  \end{description}}

\bloc{Méthodes d'enseignement (cours théoriques ; travaux de recherche et discussion en groupe ; ...)}{%
